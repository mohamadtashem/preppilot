import re

import numpy as np
import pandas as pd

def clean_text(value):
    """Trim spaces and collapse repeated spaces inside a text value."""
    if isinstance(value, str):
        return re.sub(r"\s+", " ", value).strip()
    return value


def to_number(series):
    """Turn text like '$1,200' or '15%' into numbers. Bad values become NaN."""
    text = series.astype(str).str.replace(r"[$€£¥,%\s]", "", regex=True)
    return pd.to_numeric(text, errors="coerce")


def to_date(series, dayfirst=False):
    """Parse dates. ISO dates (2024-12-31) are read as ISO; others use dayfirst."""
    raw = series.astype(str).str.strip()
    iso = raw.str.match(r"^\d{4}-\d{1,2}-\d{1,2}")
    iso_dates = pd.to_datetime(raw.where(iso), errors="coerce", format="ISO8601")
    other_dates = pd.to_datetime(raw.where(~iso), errors="coerce", format="mixed", dayfirst=dayfirst)
    return iso_dates.fillna(other_dates)


df = pd.read_csv('messy_sales.csv', sep=',', encoding='utf-8-sig')

# Step 1: Tidy column names - Renamed 12 columns to lowercase_with_underscores (e.g.
# 'Order ID' → order_id).
df = df.rename(columns={'Order ID': 'order_id',
 ' Order Date': 'order_date',
 'Customer Name': 'customer_name',
 'Region': 'region',
 'Product': 'product',
 'Category': 'category',
 'Quantity': 'quantity',
 'Unit Price': 'unit_price',
 'Discount %': 'discount',
 'Customer Rating': 'customer_rating',
 'Sales Channel': 'sales_channel',
 'Notes': 'notes'})

# Step 2: Trim text and mark placeholders as missing - Trimmed extra spaces in 292
# values and marked 36 placeholders like 'N/A' or '-' as missing.
null_tokens = ['', '-', '?', 'NA', 'N/A', 'n/a', 'na', 'null', 'NULL', 'None', 'none', 'nan',
 'NaN']
text_cols = ['order_id', 'order_date', 'customer_name', 'region', 'product', 'category',
 'unit_price', 'discount', 'customer_rating', 'sales_channel', 'notes']
for col in text_cols:
    df[col] = df[col].map(clean_text).replace(null_tokens, np.nan)

# Step 3: Drop mostly empty columns - Removed notes (94% empty).
df = df.drop(columns=['notes'])

# Step 4: Fix column types - Converted order_date → date; unit_price → number; discount
# → number; customer_rating → number.
df['order_date'] = to_date(df['order_date'], dayfirst=True)
df['unit_price'] = to_number(df['unit_price'])
df['discount'] = to_number(df['discount'])
df['customer_rating'] = to_number(df['customer_rating'])

# Step 5: Unify spelling variants - Merged 5 spelling variants in region into one
# spelling (e.g. 'east' → 'East').
df['region'] = df['region'].replace({'east': 'East',
 'NORTH': 'North',
 'north': 'North',
 'south': 'South',
 'west': 'West'})

# Step 6: Remove duplicate rows - Removed 18 duplicate rows (compared on all columns).
df = df.drop_duplicates(subset=None, keep='first').reset_index(drop=True)

# Step 7: Handle outliers - Capped 4 values beyond 3 × IQR in quantity (4).
for col in ['quantity']:
    q1, q3 = df[col].quantile([0.25, 0.75])
    spread = q3 - q1
    low, high = q1 - 3 * spread, q3 + 3 * spread
    if not spread > 0:  # (almost) all values are the same, so the rule does not apply
        continue
    if pd.api.types.is_integer_dtype(df[col]):  # keep whole numbers whole
        low, high = int(np.floor(low)), int(np.ceil(high))
    df[col] = df[col].clip(low, high)

# Step 8: Fill missing numbers - Filled 128 missing values in quantity, unit_price,
# customer_rating with each column's median.
cols = ['quantity', 'unit_price', 'customer_rating']
df[cols] = df[cols].fillna(df[cols].median())

# Step 9: Add revenue column - Created revenue = quantity * unit_price * (1 - discount /
# 100), so there is an amount to analyse.
df['revenue'] = df.eval('`quantity` * `unit_price` * (1 - `discount` / 100)')

# Step 10: Extract date parts - Added year, month and weekday columns from order_date.
df['order_date_year'] = df['order_date'].dt.year
df['order_date_month'] = df['order_date'].dt.month
df['order_date_weekday'] = df['order_date'].dt.day_name()

# Step 11: Sort rows - Sorted rows by order_date, oldest first.
df = df.sort_values(['order_date'], ascending=True).reset_index(drop=True)

df.to_csv("cleaned_data.csv", index=False)
print(f"Saved cleaned_data.csv with {len(df):,} rows and {df.shape[1]} columns.")
